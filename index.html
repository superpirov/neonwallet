            export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    const sym = url.searchParams.get('symbol') || 'ETH';
    const convert = url.searchParams.get('convert') || 'USD';
    const api = `https://pro-api.coinmarketcap.com/v1/cryptocurrency/quotes/latest?symbol=${sym}&convert=${convert}`;
    const r = await fetch(api, { headers: { 'X-CMC_PRO_API_KEY': env.CMC_KEY } });
    return new Response(r.body, { status: r.status, headers: { 'Access-Control-Allow-Origin': '*', 'Content-Type': 'application/json' }});
  }
}
